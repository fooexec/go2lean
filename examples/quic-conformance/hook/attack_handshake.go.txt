package handshake

// Copied into internal/handshake of the attacker copy by run.sh.

// AttackTransportParams, if set, rewrites the encoded transport parameters
// before they are handed to TLS.
var AttackTransportParams func([]byte) []byte

func attackTP(b []byte) []byte {
	if AttackTransportParams != nil {
		return AttackTransportParams(b)
	}
	return b
}
