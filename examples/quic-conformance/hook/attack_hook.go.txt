package quic

import (
	"github.com/quic-go/quic-go/internal/handshake"
	"github.com/quic-go/quic-go/internal/protocol"
)

// This file is copied into a scratch checkout of quic-go by run.sh. It gives
// the attacking endpoint a way to put arbitrary bytes into the next 1-RTT
// packet as one frame. The endpoint under test is an unmodified checkout.

type attackRaw []byte

func (r attackRaw) Append(b []byte, _ protocol.Version) ([]byte, error) { return append(b, r...), nil }
func (r attackRaw) Length(protocol.Version) protocol.ByteCount          { return protocol.ByteCount(len(r)) }

// AttackSendRaw queues b as one frame, bytes exactly as given.
func (c *Conn) AttackSendRaw(b []byte) {
	c.framer.QueueControlFrame(attackRaw(append([]byte(nil), b...)))
	c.scheduleSending()
}

// AttackPreSeal, if set, is called with every packet the endpoint builds,
// after the header and the payload are written and before the packet is
// encrypted and header protection is applied. raw[0] is the first byte of the
// header (all its bits are protected by header protection), raw[payloadOffset:]
// the plaintext payload. The function may change bytes in place.
var AttackPreSeal func(raw []byte, payloadOffset, pnLen int)

// AttackSetTransportParams sets a function that rewrites the encoded
// transport parameters this endpoint sends in the TLS handshake.
func AttackSetTransportParams(f func([]byte) []byte) { handshake.AttackTransportParams = f }
