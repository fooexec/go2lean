// RFC 9002 loss detection and PTO, checked against independent references.
// Times are int64 nanoseconds; a zero monotime.Time means "not set", so the
// specs draw set times from a positive range.

const specMaxPTO = int64(60 * time.Second)

func specAbs(x int64) int64 {
	if x < 0 {
		if x == -x { // math.MinInt64
			return 0
		}
		return -x
	}
	return x
}

// specScaledRef is RFC 9002 6.2.1's backoff: the PTO period doubles with each
// consecutive PTO (2^pto_count); quic-go caps it at 60 s and treats a
// non-positive period as the cap.
func specScaledRef(base int64, n uint32) int64 {
	if base <= 0 {
		return specMaxPTO
	}
	v := base
	for i := uint32(0); i < n && v <= specMaxPTO; i++ {
		v *= 2
	}
	if v > specMaxPTO {
		return specMaxPTO
	}
	return v
}

type ptoEvent struct {
	Kind uint8
	A    int64
	B    int64
}

// go2lean:sequence sentPacketHandler.getScaledPTO
// SeqScaledPTOBackoff: after any RTT history and any PTO count, the scaled PTO
// equals the RFC 9002 6.2.1 backoff (capped at 60 s), is positive, never
// above the cap, and one more consecutive PTO never shortens it.
func SeqScaledPTOBackoff(events []ptoEvent) bool {
	rtt := utils.NewRTTStats()
	h := &sentPacketHandler{rttStats: rtt}
	for _, e := range events {
		switch e.Kind % 4 {
		case 0:
			rtt.UpdateRTT(time.Duration(e.A%10_000_000_000), time.Duration(specAbs(e.B)%1_000_000_000))
		case 1:
			// max_ack_delay: values of 2^14 ms or more are invalid (RFC 9000 18.2).
			rtt.SetMaxAckDelay(time.Duration(specAbs(e.A) % (16384 * 1_000_000)))
		case 2:
			h.ptoCount = uint32(e.A)
		case 3:
			h.ptoCount++
		}
		for _, incl := range []bool{false, true} {
			got := int64(h.getScaledPTO(incl))
			if got <= 0 || got > specMaxPTO {
				return false
			}
			if got != specScaledRef(int64(rtt.PTO(incl)), h.ptoCount) {
				return false
			}
			if h.ptoCount != ^uint32(0) {
				h.ptoCount++
				next := int64(h.getScaledPTO(incl))
				h.ptoCount--
				if next < got {
					return false
				}
			}
		}
	}
	return true
}

type spaceEvent struct {
	Kind  uint8
	Space uint8
	T     int64
	N     uint8
}

func specSpace(h *sentPacketHandler, s uint8) (*packetNumberSpace, protocol.EncryptionLevel) {
	switch s % 3 {
	case 0:
		return h.initialPackets, protocol.EncryptionInitial
	case 1:
		return h.handshakePackets, protocol.EncryptionHandshake
	}
	return h.appDataPackets, protocol.Encryption1RTT
}

// specRefLossTime is RFC 9002 A.8 GetLossTimeAndSpace: the earliest set
// loss_time over the spaces still in use, the earlier space on a tie.
func specRefLossTime(h *sentPacketHandler) (monotime.Time, protocol.EncryptionLevel) {
	var t monotime.Time
	lvl := protocol.Encryption1RTT
	for s := uint8(0); s < 3; s++ {
		sp, l := specSpace(h, s)
		if sp == nil || sp.lossTime == 0 {
			continue
		}
		if t == 0 || sp.lossTime < t {
			t, lvl = sp.lossTime, l
		}
	}
	return t, lvl
}

// specRefPTO is RFC 9002 A.8 GetPtoTimeAndSpace. The anti-deadlock branch
// (no crypto packet in flight before the handshake is confirmed) arms the
// timer from now, unless the peer has completed address validation; an
// Application Data PTO is never set before the handshake is confirmed, and
// only that space adds max_ack_delay.
func specRefPTO(h *sentPacketHandler, now monotime.Time) (monotime.Time, protocol.EncryptionLevel) {
	crypto := (h.initialPackets != nil && h.initialPackets.numOutstanding > 0) ||
		(h.handshakePackets != nil && h.handshakePackets.numOutstanding > 0)
	if !h.handshakeConfirmed && !crypto {
		if h.peerCompletedAddressValidation {
			return 0, 0
		}
		t := monotime.Time(int64(now) + specScaledRef(int64(h.rttStats.PTO(false)), h.ptoCount))
		if h.initialPackets != nil {
			return t, protocol.EncryptionInitial
		}
		return t, protocol.EncryptionHandshake
	}
	var t monotime.Time
	var lvl protocol.EncryptionLevel
	for s := uint8(0); s < 3; s++ {
		sp, l := specSpace(h, s)
		if sp == nil || sp.numOutstanding == 0 || sp.lastAckElicitingPacketTime == 0 {
			continue
		}
		incl := false
		if s == 2 {
			if !h.handshakeConfirmed {
				break
			}
			incl = true
		}
		c := monotime.Time(int64(sp.lastAckElicitingPacketTime) + specScaledRef(int64(h.rttStats.PTO(incl)), h.ptoCount))
		if t == 0 || c < t {
			t, lvl = c, l
		}
	}
	return t, lvl
}

// go2lean:sequence sentPacketHandler.getPTOTimeAndSpace
// SeqLossAndPTOTimers: over reachable handshake states (Initial dropped
// before Handshake, the handshake confirmed when Handshake is dropped), the
// loss timer and the PTO timer pick the time and space RFC 9002 A.8 gives;
// no Application Data PTO before the handshake is confirmed; an outstanding
// ack-eliciting crypto packet always arms a PTO.
func SeqLossAndPTOTimers(events []spaceEvent) bool {
	rtt := utils.NewRTTStats()
	h := &sentPacketHandler{
		rttStats:         rtt,
		initialPackets:   newPacketNumberSpace(),
		handshakePackets: newPacketNumberSpace(),
		appDataPackets:   newPacketNumberSpace(),
	}
	now := monotime.Time(4_000_000_000_000)
	for _, e := range events {
		sp, _ := specSpace(h, e.Space)
		at := monotime.Time(4_000_000_000_000 + specAbs(e.T)%100_000_000_000)
		switch e.Kind % 7 {
		case 0:
			if sp != nil {
				if e.N%4 == 0 {
					sp.lossTime = 0
				} else {
					sp.lossTime = at
				}
			}
		case 1:
			if sp != nil {
				sp.lastAckElicitingPacketTime = at
			}
		case 2:
			if sp != nil {
				sp.numOutstanding = int(e.N % 4)
			}
		case 3:
			h.initialPackets = nil
		case 4:
			h.initialPackets, h.handshakePackets = nil, nil
			h.handshakeConfirmed = true
		case 5:
			h.peerCompletedAddressValidation = !h.peerCompletedAddressValidation
		case 6:
			rtt.UpdateRTT(time.Duration(1+specAbs(e.T)%1_000_000_000), 0)
			h.ptoCount = uint32(e.N % 8)
		}
		now = monotime.Time(int64(now) + int64(e.N))

		lt, ll := h.getLossTimeAndSpace()
		rt, rl := specRefLossTime(h)
		if lt != rt || (rt != 0 && ll != rl) {
			return false
		}
		pt, pl := h.getPTOTimeAndSpace(now)
		qt, ql := specRefPTO(h, now)
		if pt != qt || (qt != 0 && pl != ql) {
			return false
		}
		if pt != 0 && pl == protocol.Encryption1RTT && !h.handshakeConfirmed {
			return false
		}
		if ((h.initialPackets != nil && h.initialPackets.numOutstanding > 0 && h.initialPackets.lastAckElicitingPacketTime != 0) ||
			(h.handshakePackets != nil && h.handshakePackets.numOutstanding > 0 && h.handshakePackets.lastAckElicitingPacketTime != 0)) && pt == 0 {
			return false
		}
	}
	return true
}

type connEvent struct {
	Kind uint8
	A    uint16
	B    uint16
	C    uint8
	D    uint8
}

func specBytesInFlight(h *sentPacketHandler) protocol.ByteCount {
	var n protocol.ByteCount
	for _, p := range h.hist.Packets() {
		if p.includedInBytesInFlight {
			n += p.Length
		}
	}
	return n
}

func specInHistory(h *sentPacketHandler, pn protocol.PacketNumber) bool {
	for qn := range h.hist.Packets() {
		if qn == pn {
			return true
		}
	}
	return false
}

// specRanges turns a set of packet numbers into ACK ranges, largest first,
// merging consecutive numbers (RFC 9000 19.3).
func specRanges(pns []protocol.PacketNumber) []wire.AckRange {
	var rs []wire.AckRange
	for _, pn := range pns { // pns is descending
		if n := len(rs); n > 0 && rs[n-1].Smallest == pn+1 {
			rs[n-1].Smallest = pn
			continue
		}
		rs = append(rs, wire.AckRange{Smallest: pn, Largest: pn})
	}
	return rs
}

const specMaxAckDelay = 25 * time.Millisecond

// go2lean:sequence sentPacketHandler.ReceivedAck
// SeqSentPacketHandler1RTT drives a server's handler after the handshake
// (address validated, Initial and Handshake dropped, ECN on, max_ack_delay
// 25 ms) through 1-RTT packets sent with the ECN marking it asks for, ACKs
// from an honest peer (any subset of the recent packets, several ranges, an
// ack delay, ECN counts of what arrived, some arrivals CE-marked by the path),
// ACKs with tampered ECN counts, ACKs of arbitrary packet number ranges, and
// loss detection timeouts. After every step:
//   - bytes_in_flight is the sum over the packets the history counts in
//     flight and never exceeds the bytes sent; the loss detection timer is
//     armed exactly while an ack-eliciting packet is outstanding (RFC 9002
//     6.2.1); a PTO timeout raises the PTO count by one;
//   - an ACK of packets that were sent is not an error, and one of a packet
//     number never sent is (RFC 9000 13.1);
//   - after an ACK that newly acknowledges a packet (RFC 9002 A.7): the PTO
//     count is 0; no in-flight packet below the largest acknowledged is older
//     than the time threshold (6.1.2) or, unless quic-go turned the packet
//     threshold off after a spurious loss, has 3 sent packets after it up to
//     the largest acknowledged (6.1.1); if the largest acknowledged is newly
//     acknowledged, not older than the last RTT sample's packet and sent
//     before this ACK arrived, the latest
//     RTT is its sample, less the ack delay capped at max_ack_delay when that
//     keeps it at least min_rtt (5.3);
//   - ECN (RFC 9000 13.4.2.1): packets are sent unmarked or ECT(0); once
//     validation failed it stays failed and packets go unmarked; an ACK that
//     does not increase the largest acknowledged does not change the
//     validation state (other than through packets it declares lost); an ACK
//     that increases it fails validation when it reports more ECT(0) than were
//     sent, acknowledges ECN-marked packets without counts, lowers a count, or
//     raises ECT(0)+ECN-CE by less than the ECT(0) packets it newly
//     acknowledges; an honest path that never CE-marks and loses no testing
//     packet never fails it;
//   - the time threshold stays within 9/8 and 16/8 and never tightens, and a
//     packet threshold turned off stays off (RFC 9002 6.1 allows relaxing
//     them after spurious losses).
func SeqSentPacketHandler1RTT(events []connEvent) bool {
	rtt := utils.NewRTTStats()
	rtt.SetMaxAckDelay(specMaxAckDelay)
	h := NewSentPacketHandler(0, 1200, rtt, &utils.ConnectionStats{}, true, true,
		func(protocol.PacketNumber) {}, protocol.PerspectiveServer, nil, utils.DefaultLogger).(*sentPacketHandler)
	t := monotime.Time(4_000_000_000_000)
	h.DropPackets(protocol.EncryptionInitial, t)
	h.DropPackets(protocol.EncryptionHandshake, t)
	ecn := h.ecnTracker.(*ecnTracker)

	var sent []protocol.PacketNumber
	isSent := map[protocol.PacketNumber]bool{}
	sendTime := map[protocol.PacketNumber]monotime.Time{}
	marking := map[protocol.PacketNumber]protocol.ECN{}
	ceMarked := map[protocol.PacketNumber]bool{}
	arrived := map[protocol.PacketNumber]bool{}
	var sentECT0, gotECT0, gotECT1, gotCE int64
	dishonest, anyCE := false, false
	var accECT0, accECT1, accCE int64 // the counts of the last ACK ECN validation accepted
	eighths, ptDisabled := h.timeThresholdEighths, h.packetThresholdDisabled

	for _, e := range events {
		t = monotime.Time(int64(t) + int64(e.A)*100_000) // up to ~6.5 s per step
		switch e.Kind % 5 {
		case 0: // send one packet, or a burst of up to 12
			burst := 1
			if e.D%3 == 0 {
				burst = 1 + int(e.C%12)
			}
			for range burst {
				failedBefore := ecn.state == ecnStateFailed
				m := h.ECNMode(true)
				if m != protocol.ECNNon && m != protocol.ECT0 {
					return false
				}
				if failedBefore && m != protocol.ECNNon {
					return false
				}
				pn := h.PopPacketNumber(protocol.PacketNumberSpaceAppData)
				h.SentPacket(t, pn, protocol.InvalidPacketNumber, nil, []Frame{{Frame: &wire.PingFrame{}}},
					protocol.Encryption1RTT, m, protocol.ByteCount(1+e.B%1400), false, false)
				sent = append(sent, pn)
				isSent[pn], sendTime[pn], marking[pn] = true, t, m
				if m == protocol.ECT0 {
					sentECT0++
					if e.D%8 == 0 { // the path marks this one CE
						ceMarked[pn] = true
					}
				}
			}
		case 1, 2: // an honest peer's ACK, or one with tampered ECN counts
			if len(sent) == 0 {
				continue
			}
			top := len(sent) - 1 - int(e.B)%len(sent)
			mask := e.C
			if mask == 0 {
				mask = 1
			}
			var pns []protocol.PacketNumber
			for k := 0; k < 8 && top-k >= 0; k++ {
				if mask&(1<<k) != 0 {
					pns = append(pns, sent[top-k])
				}
			}
			if len(pns) == 0 {
				pns = append(pns, sent[top])
			}
			for _, pn := range pns {
				if !arrived[pn] {
					arrived[pn] = true
					switch {
					case ceMarked[pn]:
						gotCE++
						anyCE = true
					case marking[pn] == protocol.ECT0:
						gotECT0++
					case marking[pn] == protocol.ECT1:
						gotECT1++
					}
				}
			}
			ect0, ect1, ce := gotECT0, gotECT1, gotCE
			overReport := false
			if e.Kind%5 == 2 {
				dishonest = true
				switch e.D % 4 {
				case 0:
					ect0 = sentECT0 + 1 + int64(e.D>>2)
					overReport = true
				case 1:
					ect0 = 0
				case 2:
					ce += int64(e.D >> 2)
				case 3:
					ect0, ect1, ce = 0, 0, 0
				}
			}
			newly := false
			var ackedECT0 int64 // newly acknowledged packets sent ECT(0)
			for _, pn := range pns {
				if specInHistory(h, pn) {
					newly = true
					if marking[pn] == protocol.ECT0 {
						ackedECT0++
					}
				}
			}
			largest := pns[0]
			largestNewly := specInHistory(h, largest)
			prevLargestTime := h.largestAckedTime
			prevLargestAcked := h.appDataPackets.largestAcked
			stateBefore, lostTestingBefore := ecn.state, ecn.numLostTesting
			failedBefore := ecn.state == ecnStateFailed
			delay := time.Duration(e.D) * time.Millisecond
			ack := &wire.AckFrame{AckRanges: specRanges(pns), DelayTime: delay,
				ECT0: uint64(ect0), ECT1: uint64(ect1), ECNCE: uint64(ce)}
			if _, err := h.ReceivedAck(ack, protocol.Encryption1RTT, t); err != nil {
				return false
			}
			if failedBefore && ecn.state != ecnStateFailed {
				return false
			}
			increases := prevLargestAcked == protocol.InvalidPacketNumber || largest > prevLargestAcked
			if !increases && ecn.numLostTesting == lostTestingBefore && ecn.state != stateBefore {
				return false
			}
			if increases && newly && !failedBefore {
				// RFC 9000 13.4.2.1: validation fails on more ECT(0) than sent,
				// on ECN-marked packets acknowledged without counts, and when
				// the increase of ECT(0) and ECN-CE is less than the newly
				// acknowledged ECT(0) packets.
				mustFail := overReport ||
					(ackedECT0 > 0 && ect0 == 0 && ect1 == 0 && ce == 0) ||
					(ect0-accECT0)+(ce-accCE) < ackedECT0 ||
					ect0 < accECT0 || ect1 < accECT1 || ce < accCE
				if mustFail && ecn.state != ecnStateFailed {
					return false
				}
				if ecn.state != ecnStateFailed {
					accECT0, accECT1, accCE = ect0, ect1, ce
				}
			}
			if !newly {
				break
			}
			if h.ptoCount != 0 {
				return false
			}
			// A sample of 0 is not taken (RTTStats ignores it; a zero RTT cannot happen).
			if sd := time.Duration(int64(t) - int64(sendTime[largest])); largestNewly && sd > 0 &&
				(prevLargestTime == 0 || sendTime[largest] >= prevLargestTime) {
				ad := min(delay, specMaxAckDelay)
				want := sd
				if sd-rtt.MinRTT() >= ad {
					want = sd - ad
				}
				if rtt.LatestRTT() != want {
					return false
				}
			}
			lossDelay := max(max(rtt.LatestRTT(), rtt.SmoothedRTT())*time.Duration(8+h.timeThresholdEighths)/8, protocol.TimerGranularity)
			la := h.appDataPackets.largestAcked
			for qn, p := range h.hist.Packets() {
				if qn >= la {
					break
				}
				if !p.includedInBytesInFlight {
					continue
				}
				if int64(p.SendTime) <= int64(t)-int64(lossDelay) {
					return false
				}
				if !h.packetThresholdDisabled {
					after := 0
					for _, s := range sent {
						if s > qn && s <= la {
							after++
						}
					}
					if after >= packetThreshold {
						return false
					}
				}
			}
		case 3: // an ACK of an arbitrary range of packet numbers
			if len(sent) == 0 {
				continue
			}
			largestSent := h.appDataPackets.largestSent
			lo := protocol.PacketNumber(int64(e.B) % (int64(largestSent) + 2))
			hi := lo + protocol.PacketNumber(e.C%4)
			allSent := true
			for pn := lo; pn <= hi; pn++ {
				if !isSent[pn] {
					allSent = false
				}
			}
			ack := &wire.AckFrame{AckRanges: []wire.AckRange{{Smallest: lo, Largest: hi}}}
			_, err := h.ReceivedAck(ack, protocol.Encryption1RTT, t)
			if hi > largestSent && err == nil {
				return false
			}
			if allSent && err != nil {
				return false
			}
			if err != nil {
				return true // the connection closes with PROTOCOL_VIOLATION
			}
			for pn := lo; pn <= hi; pn++ {
				if isSent[pn] && !arrived[pn] {
					// the peer claims it: count it as arrived for honest counts
					arrived[pn] = true
					switch {
					case ceMarked[pn]:
						gotCE++
						anyCE = true
					case marking[pn] == protocol.ECT0:
						gotECT0++
					}
				}
			}
			dishonest = true // no ECN counts on this ACK
		case 4: // the loss detection timer fires
			at := h.GetLossDetectionTimeout()
			if at == 0 {
				continue
			}
			if at > t {
				t = at
			}
			before := h.ptoCount
			pto := h.alarm.TimerType == qlog.TimerTypePTO
			if err := h.OnLossDetectionTimeout(t); err != nil {
				return false
			}
			if pto && h.ptoCount != before+1 {
				return false
			}
		}
		if h.bytesInFlight != specBytesInFlight(h) || h.bytesInFlight > h.bytesSent {
			return false
		}
		if h.hist.HasOutstandingPackets() == (h.GetLossDetectionTimeout() == 0) {
			return false
		}
		if !dishonest && !anyCE && ecn.numLostTesting < ecn.numSentTesting && ecn.state == ecnStateFailed {
			return false
		}
		// After spurious losses quic-go may relax its thresholds (RFC 9002
		// 6.1.1, 6.1.2 MAY): the time threshold stays within 9/8 and 16/8
		// and never tightens; the packet threshold, once off, stays off.
		if h.timeThresholdEighths < eighths || h.timeThresholdEighths < 1 || h.timeThresholdEighths > 8 ||
			(ptDisabled && !h.packetThresholdDisabled) {
			return false
		}
		eighths, ptDisabled = h.timeThresholdEighths, h.packetThresholdDisabled
	}
	return true
}
