package ackhandler

import (
	"time"

	"github.com/quic-go/quic-go/internal/protocol"
	"github.com/quic-go/quic-go/internal/utils"
	"github.com/quic-go/quic-go/internal/wire"
)

type calHandler struct{}

func (calHandler) OnAcked(wire.Frame) {}
func (calHandler) OnLost(wire.Frame)  {}

// SeqSentPacketHandlerPathProbesNoPanic checks candidate c001: whatever
// the order of sent packets, path probes, ACKs and loss detection
// timeouts, the sent packet handler does not panic; an ACK it cannot
// process is an error. Each op: bits 0-1 pick the action (send, send a
// path probe, loss detection timeout, ACK), bits 2-3 how far the clock
// moves after it, bits 4-11 which of the first 8 packets the ACK names.
//
// go2lean:sequence sentPacketHandler.ReceivedAck
func SeqSentPacketHandlerPathProbesNoPanic(ops []uint16) bool {
	if len(ops) > 8 {
		ops = ops[:8]
	}
	var rtt utils.RTTStats
	rtt.UpdateRTT(100*time.Millisecond, 0)
	h := newSentPacketHandler(0, 1200, &rtt, true, false, protocol.PerspectiveClient, nil, utils.DefaultLogger)
	now := time.Unix(1000, 0)
	h.DropPackets(protocol.EncryptionInitial, now)
	h.DropPackets(protocol.EncryptionHandshake, now)
	steps := [4]time.Duration{0, 100 * time.Millisecond, 500 * time.Millisecond, 1500 * time.Millisecond}
	var pns []protocol.PacketNumber
	for _, op := range ops {
		switch op & 3 {
		case 0, 1:
			pn := h.PopPacketNumber(protocol.Encryption1RTT)
			h.SentPacket(now, pn, protocol.InvalidPacketNumber, nil, []Frame{{Frame: &wire.PingFrame{}, Handler: calHandler{}}},
				protocol.Encryption1RTT, protocol.ECNNon, 1200, false, op&3 == 1)
			pns = append(pns, pn)
		case 2:
			h.OnLossDetectionTimeout(now)
		case 3:
			var acked []protocol.PacketNumber
			for i, pn := range pns {
				if i < 8 && op>>(4+uint(i))&1 == 1 {
					acked = append(acked, pn)
				}
			}
			if len(acked) > 0 {
				// The ranges, largest first, each of consecutive numbers.
				var ranges []wire.AckRange
				for i := len(acked) - 1; i >= 0; i-- {
					if n := len(ranges); n > 0 && ranges[n-1].Smallest == acked[i]+1 {
						ranges[n-1].Smallest = acked[i]
					} else {
						ranges = append(ranges, wire.AckRange{Smallest: acked[i], Largest: acked[i]})
					}
				}
				h.ReceivedAck(&wire.AckFrame{AckRanges: ranges}, protocol.Encryption1RTT, now)
			}
		}
		now = now.Add(steps[op>>2&3])
	}
	h.OnLossDetectionTimeout(now)
	return true
}
