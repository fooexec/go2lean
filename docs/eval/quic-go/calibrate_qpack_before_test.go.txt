package http3

import (
	"io"
	"strconv"
	"strings"

	"github.com/quic-go/qpack"
)

// SeqRequestHeaderSizeBounded checks candidate c004: a request whose
// decoded field section is larger than the configured limit is rejected.
// RFC 9114, Section 4.2.2: the size is the length of each field's name and
// value plus 32. The request carries its four pseudo-header fields and n
// regular fields of vlen bytes each; the limit is 1024.
//
// go2lean:sequence requestFromHeaders
func SeqRequestHeaderSizeBounded(n, vlen uint8) bool {
	const limit = 1024
	fields := []qpack.HeaderField{
		{Name: ":method", Value: "GET"},
		{Name: ":scheme", Value: "https"},
		{Name: ":authority", Value: "a"},
		{Name: ":path", Value: "/"},
	}
	value := strings.Repeat("a", int(vlen%128))
	for i := 0; i < int(n%64); i++ {
		fields = append(fields, qpack.HeaderField{Name: "x-h" + strconv.Itoa(i), Value: value})
	}
	size := 0
	for _, f := range fields {
		size += len(f.Name) + len(f.Value) + 32
	}
	next := 0
	dec := func() (qpack.HeaderField, error) {
		if next == len(fields) {
			return qpack.HeaderField{}, io.EOF
		}
		next++
		return fields[next-1], nil
	}
	_, err := requestFromHeaders(dec, nil)
	return size <= limit || err != nil
}
