package quic

import "github.com/quic-go/quic-go/internal/protocol"

// SeqDataBlockedReported checks candidate c091 (issue #4174): when the
// connection's send credit is exhausted, including when the peer's
// initial_max_data is 0, the sender reports it, once per limit, with the
// limit it is blocked at (RFC 9000, Section 4.1: "A sender SHOULD send a
// DATA_BLOCKED frame ... when it wishes to send data but is unable to do
// so due to connection-level flow control"). Events: send bytes within
// the credit, raise the limit (MAX_DATA), ask whether newly blocked.
//
// go2lean:sequence connectionFlowController.IsNewlyBlocked
func SeqDataBlockedReported(initialMaxData uint16, events []uint16) bool {
	c := newConnectionFlowController(1<<20, 1<<20, nil, nil, nil)
	window := protocol.ByteCount(initialMaxData % 512)
	c.UpdateSendWindow(window)
	var sent protocol.ByteCount
	reported := map[protocol.ByteCount]bool{}
	for _, e := range events {
		arg := protocol.ByteCount(e >> 2)
		switch e % 3 {
		case 0:
			if n := min(arg%300, window-sent); c.TryAddBytesSent(n) {
				sent += n
			}
		case 1:
			if lim := arg % 1024; c.UpdateSendWindow(lim) {
				window = lim
			}
		case 2:
			blocked, at := c.IsNewlyBlocked()
			if blocked {
				if at != window || reported[window] || sent < window {
					return false
				}
				reported[window] = true
			} else if sent == window && !reported[window] {
				// Blocked at this limit and never reported.
				return false
			}
		}
	}
	return true
}
